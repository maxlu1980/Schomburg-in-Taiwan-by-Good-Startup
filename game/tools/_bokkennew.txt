function bokken() { if (FSV.bokken) return FSV.bokken; /* (Köln, V62) the pioneer's spade: a short ash shaft with a D-grip, a field-grey steel blade, slightly dished (she strikes with its flat) */
    var g = new THREE.Group(), wood = LUNA.toon(0x9a7448, { rim: 0.15 }), steel = LUNA.toon(0x8a9096, { rim: 0.35 }), b = LUNA.mesh(new THREE.CylinderGeometry(0.012, 0.015, 0.3, 14), wood, { ink: 0.0025 }); b.rotation.x = Math.PI / 2; b.position.z = 0.15; g.add(b);
    var grip = LUNA.mesh(new THREE.TorusGeometry(0.026, 0.007, 10, 22), wood, { ink: 0.002 }); grip.rotation.y = Math.PI / 2; grip.position.z = -0.024; g.add(grip);
    var bg = new THREE.CylinderGeometry(0.085, 0.085, 0.14, 18, 6, true, -0.72, 1.44), pa = bg.attributes.position; for (var i = 0; i < pa.count; i++) { var u = Math.min(1, Math.max(0, (pa.getY(i) + 0.07) / 0.14)), k = 1 - 0.5 * Math.pow(u, 2.4); pa.setX(i, pa.getX(i) * k); pa.setZ(i, (pa.getZ(i) - 0.085) * (0.6 + 0.4 * k)); }
    bg.translate(0, 0.07, 0); bg.rotateX(Math.PI / 2); bg.computeVertexNormals(); var bl = new THREE.Mesh(bg, LUNA.toon(0x8a9096, { rim: 0.4, side: THREE.DoubleSide })); bl.castShadow = true; bl.position.z = 0.3; g.add(bl);
    var col = LUNA.mesh(new THREE.CylinderGeometry(0.017, 0.013, 0.05, 14), steel, { ink: 0.002 }); col.rotation.x = Math.PI / 2; col.position.z = 0.29; g.add(col);
    g.position.set(0, -0.04, 0.0); g.visible = false; luna.armR.wr.add(g); FSV.bokken = g; return g; }
